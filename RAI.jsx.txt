function Toggle({ on, onToggle }) {
  return (
    <div className={`toggle ${on ? "on" : ""}`} onClick={onToggle}>
      <div className="toggle-knob" />
    </div>
  );
}

const WELCOME = "Initializing neural interface… Hello! I'm RAI, your AI companion. How can I assist you today?";

// ─── Main App ────────────────────────────────────────────────────────────────
export default function RAI() {
  const [messages, setMessages] = useState([
    { role: "ai", text: WELCOME, id: 0 }
  ]);
  const [input, setInput]           = useState("");
  const [status, setStatus]         = useState("online"); // online | listening | speaking | thinking
  const [typing, setTyping]         = useState(false);
  const [showSettings, setSettings] = useState(false);
  const [micActive, setMicActive]   = useState(false);
  const [toast, setToast]           = useState(null);
  const [settings, setSettingsVal]  = useState({
    voice: true, autoSpeak: true, soundFx: false,
    voiceGender: "female", historyEnabled: true,
  });
  const [history, setHistory]       = useState([]);   // for Claude API

  const chatRef   = useRef(null);
  const inputRef  = useRef(null);
  const synthRef  = useRef(window.speechSynthesis);
  const recogRef  = useRef(null);
  const idRef     = useRef(1);
  const abortRef  = useRef(null);

  // Scroll to bottom
  useEffect(() => {
    chatRef.current?.scrollTo({ top: chatRef.current.scrollHeight, behavior: "smooth" });
  }, [messages, typing]);

  // Speak with TTS
  const speak = useCallback((text) => {
    if (!settings.voice || !settings.autoSpeak) return;
    synthRef.current?.cancel();
    const utt = new SpeechSynthesisUtterance(text.replace(/[*_`#]/g, ""));
    utt.rate = 1.05; utt.pitch = settings.voiceGender === "female" ? 1.2 : 0.85;
    const voices = synthRef.current.getVoices();
    const pick = settings.voiceGender === "female"
      ? voices.find(v => /female|samantha|zira|karen|victoria|google uk english female/i.test(v.name))
      : voices.find(v => /male|alex|daniel|david|google uk english male/i.test(v.name));
    if (pick) utt.voice = pick;
    utt.onstart = () => setStatus("speaking");
    utt.onend   = () => setStatus("online");
    utt.onerror = () => setStatus("online");
    synthRef.current.speak(utt);
  }, [settings.voice, settings.autoSpeak, settings.voiceGender]);

  // Stop speech
  const stopSpeech = () => { synthRef.current?.cancel(); setStatus("online"); };

  // Send message to Claude API
  const sendMessage = useCallback(async (userText) => {
    if (!userText.trim()) return;
    const uid = idRef.current++;
    const newUserMsg = { role: "user", text: userText, id: uid };
    const newHistory = [...history, { role: "user", content: userText }];

    setMessages(prev => [...prev, newUserMsg]);
    setHistory(newHistory);
    setInput(""); setTyping(true); setStatus("thinking");
    stopSpeech();

    const controller = new AbortController();
    abortRef.current = controller;

    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        signal: controller.signal,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          system: "You are RAI, a friendly, intelligent, and slightly futuristic AI companion. Respond naturally and warmly. Keep answers concise (2-4 sentences for casual talk, more for technical topics). Use a conversational tone. Do NOT use markdown formatting like ** or ## in your responses.",
          messages: newHistory,
        }),
      });

      const data = await res.json();
      const aiText = data?.content?.[0]?.text || "I encountered a glitch in my neural net. Could you try again?";

      setTyping(false);
      const aid = idRef.current++;
      setMessages(prev => [...prev, { role: "ai", text: "", id: aid, fresh: true }]);
      setHistory(prev => [...prev, { role: "assistant", content: aiText }]);

      // Typewriter effect
      setStatus("speaking");
      let i = 0;
      const tick = setInterval(() => {
        i++;
        setMessages(prev => prev.map(m => m.id === aid ? { ...m, text: aiText.slice(0, i) } : m));
        if (i >= aiText.length) {
          clearInterval(tick);
          setMessages(prev => prev.map(m => m.id === aid ? { ...m, fresh: false } : m));
          speak(aiText);
          if (!settings.voice || !settings.autoSpeak) setStatus("online");
        }
      }, 18);

    } catch (e) {
      if (e.name !== "AbortError") {
        setTyping(false); setStatus("online");
        const eid = idRef.current++;
        setMessages(prev => [...prev, { role: "ai", text: "Connection lost. Please try again.", id: eid }]);
      }
    }
  }, [history, speak, settings]);

  // Handle key
  const handleKey = (e) => {
    if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); sendMessage(input); }
  };

  // Mic toggle
  const toggleMic = () => {
    if (!("webkitSpeechRecognition" in window || "SpeechRecognition" in window)) {
      showToast("🎙 Speech recognition not supported in this browser"); return;
    }
    if (micActive) {
      recogRef.current?.stop();
      setMicActive(false); setStatus("online"); return;
    }
    const SRec = window.SpeechRecognition || window.webkitSpeechRecognition;
    const rec = new SRec();
    rec.continuous = false; rec.interimResults = false; rec.lang = "en-US";
    rec.onstart   = () => { setMicActive(true); setStatus("listening"); };
    rec.onresult  = (e) => {
      const t = e.results[0][0].transcript;
      setMicActive(false); setStatus("online");
      sendMessage(t);
    };
    rec.onerror = rec.onend = () => { setMicActive(false); setStatus("online"); };
    recogRef.current = rec;
    rec.start();
  };

  const showToast = (msg) => { setToast(msg); setTimeout(() => setToast(null), 2600); };

  const clearHistory = () => {
    if (!settings.historyEnabled) return;
    setMessages([{ role: "ai", text: WELCOME, id: 0 }]);
    setHistory([]);
    setSettings(false);
    showToast("✓ Conversation cleared");
  };

  const statusLabel = { online: "online", listening: "listening", speaking: "speaking", thinking: "processing" }[status] || "online";

  return (
    <>
      <style>{CSS}</style>
      <div className="grid-bg" />
      <div className="nebula" />

      <div className="app">
        {/* Header */}
        <header className="header">
          <div className="logo">
            <div className="logo-mark">RAI</div>
            <div>
              <div className="logo-title">RAI</div>
              <div className="logo-sub">Your AI Companion</div>
            </div>
          </div>
          <div className="header-actions">
            <button className="icon-btn" onClick={() => setSettings(true)} title="Settings">⚙</button>
            <button className="icon-btn" onClick={stopSpeech} title="Stop speech">⏹</button>
          </div>
        </header>

        {/* Avatar */}
        <div className="avatar-panel">
          <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 8 }}>
            <div className={`avatar-wrap ${status === "speaking" ? "speaking" : ""}`}>
              <div className="avatar-ring" />
              <div className="avatar-ring r2" />
              <div className="avatar-ring r3" />
              <div className="avatar-core">
                <div className="avatar-face">
                  <div className="eye left" />
                  <div className="eye right" />
                  <div className={`mouth ${status === "speaking" ? "speaking" : ""}`} />
                  <div className="scanline" />
                  <div className="glitch-bar" />
                </div>
              </div>
            </div>
            <div className="status-pill">
              <div className={`status-dot ${status === "listening" ? "listening" : status === "speaking" ? "speaking" : ""}`} />
              {statusLabel}
            </div>
          </div>
        </div>

        {/* Chat */}
        <div className="chat-area" ref={chatRef}>
          {messages.map(m => (
            <div key={m.id} className={`msg-row ${m.role}`}>
              <div className={`bubble ${m.role} ${m.fresh ? "fresh" : ""}`}>
                {m.role === "ai" && <div className="msg-label">RAI</div>}
                {m.role === "user" && <div className="msg-label" style={{ textAlign: "right" }}>You</div>}
                <span>{m.text}</span>
                {m.fresh && m.text.length > 0 && <span className="cursor" />}
              </div>
            </div>
          ))}
          {typing && (
            <div className="msg-row ai">
              <div className="bubble ai">
                <div className="msg-label">RAI</div>
                <div className="typing-indicator">
                  <span /><span /><span />
                </div>
              </div>
            </div>
          )}
        </div>

        {/* Input */}
        <div className="input-bar">
          <button className={`mic-btn ${micActive ? "active" : ""}`} onClick={toggleMic} title="Voice input">
            🎙
          </button>
          <div className="input-wrap">
            <textarea
              ref={inputRef}
              className="chat-input"
              value={input}
              onChange={e => setInput(e.target.value)}
              onKeyDown={handleKey}
              placeholder="Send a message to RAI…"
              rows={1}
              style={{ height: "auto" }}
              onInput={e => {
                e.target.style.height = "auto";
                e.target.style.height = Math.min(e.target.scrollHeight, 130) + "px";
              }}
            />
          </div>
          <button className="send-btn" onClick={() => sendMessage(input)} disabled={!input.trim() || typing}>
            ➤
          </button>
        </div>
      </div>

      {/* Settings */}
      {showSettings && (
        <div className="overlay" onClick={e => e.target === e.currentTarget && setSettings(false)}>
          <div className="settings-panel">
            <div className="panel-header">
              <div className="panel-title">// SETTINGS</div>
              <button className="icon-btn" onClick={() => setSettings(false)}>✕</button>
            </div>

            {[
              { key: "voice",          label: "Voice Output",     hint: "RAI speaks responses aloud" },
              { key: "autoSpeak",      label: "Auto-Speak",       hint: "Speak each reply automatically" },
              { key: "historyEnabled", label: "Chat History",     hint: "Keep conversation context" },
            ].map(s => (
              <div className="setting-row" key={s.key}>
                <div className="setting-label">
                  {s.label}
                  <span className="setting-hint">{s.hint}</span>
                </div>
                <Toggle on={settings[s.key]} onToggle={() => setSettingsVal(p => ({ ...p, [s.key]: !p[s.key] }))} />
              </div>
            ))}

            <div className="setting-row">
              <div className="setting-label">
                Voice Gender
                <span className="setting-hint">TTS voice preference</span>
              </div>
              <select
                className="setting-select"
                value={settings.voiceGender}
                onChange={e => setSettingsVal(p => ({ ...p, voiceGender: e.target.value }))}
              >
                <option value="female">Female</option>
                <option value="male">Male</option>
              </select>
            </div>

            <button className="danger-btn" onClick={clearHistory}>
              ⚠ Clear Conversation
            </button>
          </div>
        </div>
      )}

      {toast && <div className="toast">{toast}</div>}
    </>
  );
}
